void Unity_Dither_float(float4 ScreenPosition, out float Out)
{
    float2 uv = ScreenPosition.xy * _ScreenParams.xy;
    float DITHER_THRESHOLDS[16] =
    {
        1.0 / 17.0,  9.0 / 17.0,  3.0 / 17.0, 11.0 / 17.0,
        13.0 / 17.0,  5.0 / 17.0, 15.0 / 17.0,  7.0 / 17.0,
        4.0 / 17.0, 12.0 / 17.0,  2.0 / 17.0, 10.0 / 17.0,
        16.0 / 17.0,  8.0 / 17.0, 14.0 / 17.0,  6.0 / 17.0
    };
    uint index = (uint(uv.x) % 4) * 4 + uint(uv.y) % 4;
    Out = 1 - DITHER_THRESHOLDS[index];
}

                float3 posObj = TransformObjectToWorld(float3(0, 0, 0));

然后，如果需要两个点作为消失点的话

void DitherClip(Varyings input)
{
half dither = Dither4x4Bayer(fmod(input.positionCS.x, 4), fmod(input.positionCS.y, 4));
half clipValue = smoothstep(0, _DitherAmount, (distance(input.positionWSAndFogFactor.xz, _WorldSpaceCameraPos.xz) - dither));
// 1和2是为了适配两个固定的消失点，用min取最小值。此时上面的clipValue是失效的
half clipValue1 = smoothstep(0, _DitherAmount, (distance(_CarDitherPos.xy, input.positionWSAndFogFactor.xz) - dither));
half clipValue2 = smoothstep(0, _DitherAmount, (distance(_CarDitherPos.zw, input.positionWSAndFogFactor.xz) - dither));
clipValue = min(clipValue1,clipValue2);
clip(clipValue - 1);
}
